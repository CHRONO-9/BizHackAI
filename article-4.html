こんにちは。「仕事効率化・生成AIツール特化メディア」エンジニア・高度自動化担当ライターです。

エンジニアや社内SE、テック系ビジネスパーソンにとって、日々の最大の生産性キラーは「コミュニケーションツール間のコンテキストスイッチ」と「散らばったタスクの手動転記」ではないでしょうか。

本記事では、サーバー構築不要・維持コスト実質ゼロ（Googleの無料枠内で稼働）のGoogle Apps Script (GAS)をオーケストレーターとし、現行LLMで最高峰の日本語推論力・構造化データ抽出精度を誇るAnthropicの「Claude 3.5 Sonnet」、そして情報ハブであるNotion APIを直結した「自律型タスク抽出パイプライン」を構築します。

2. システム全体像と処理シーケンス

本システムは、GASの時間主導型トリガー（Time-driven Trigger）によって10分おきにバックグラウンド自律起動します。

  1. 差分検知：Gmailの未読メールをクエリ検索（`label:UNREAD -label:AI-Processed`）。
  2. LLM推論：Claude 3.5 Sonnet の Tool Use（Structured Outputs）で厳格なJSONタスク構造体を抽出。「来週水曜」「月末」といった相対日付を、実行日を基準とした絶対日付（YYYY-MM-DD）に変換。
  3. Notion登録：Message IDに基づく重複チェック（冪等性の担保）を行い、Notion DBへUPSERT。処理済みラベルを付与。

4. 【完全掲載】Google Apps Scriptの実装コード（Code.gs）

GASのプロジェクト内に貼り付けてそのまま動くプロダクションコードです。 GAS特有の「6分制限」に対するセーフティ中断と、APIレートリミット（429）を自動リトライする指数バックオフを標準装備しています。

Google Apps Script (Code.gs) 実装コード 
/**
 * 自律型タスク抽出パイプライン: Gmail -> Claude 3.5 Sonnet -> Notion
 */
const EXECUTION_TIMEOUT_MS = 270 * 1000; // 4分30秒でセーフティ中断

function getEnv(key) {
  const val = PropertiesService.getScriptProperties().getProperty(key);
  if (!val) throw new Error(`[Config Error] "${key}" が設定されていません。`);
  return val;
}

function main() {
  const startTime = Date.now();
  console.log(`[Start] タスク抽出パイプラインを開始: ${new Date().toISOString()}`);
  try {
    processGmailPipeline(startTime);
    console.log(`[Finish] 終了。経過時間: ${(Date.now() - startTime) / 1000}s`);
  } catch (error) {
    console.error(`[Error] 異常終了: ${error.stack}`);
    sendSlackAlert(`:rotating_light: *[GASパイプライン異常終了]*\n\`\`\`${error.stack}\`\`\``);
  }
}

function processGmailPipeline(startTime) {
  const query = PropertiesService.getScriptProperties().getProperty('GMAIL_SEARCH_QUERY') || 'is:unread -label:AI-Processed';
  const threads = GmailApp.search(query, 0, 15);
  if (threads.length === 0) return;

  let processedLabel = GmailApp.getUserLabelByName('AI-Processed') || GmailApp.createLabel('AI-Processed');

  for (const thread of threads) {
    if (Date.now() - startTime > EXECUTION_TIMEOUT_MS) {
      console.warn('[Safety] 実行時間上限に近づいたため中断します。');
      break;
    }

    const messages = thread.getMessages();
    const lastMessage = messages[messages.length - 1];
    const messageId = lastMessage.getId();
    const subject = thread.getFirstMessageSubject();
    const sender = lastMessage.getFrom();
    const body = lastMessage.getPlainBody().substring(0, 4000);
    const threadUrl = `https://mail.google.com/mail/u/0/#inbox/${thread.getId()}`;

    const result = extractTasksWithClaude({
      source: 'Gmail',
      title: subject,
      sender: sender,
      content: body
    });

    if (result && result.has_task && result.tasks.length > 0) {
      for (const task of result.tasks) {
        createNotionTask({
          title: task.task_name,
          dueDate: task.due_date,
          priority: task.priority,
          sourceType: 'Gmail',
          sender: sender,
          summary: task.summary,
          externalId: `${messageId}_${task.task_name}`,
          sourceUrl: threadUrl
        });
      }
    }
    thread.addLabel(processedLabel);
    thread.markRead();
  }
}

function extractTasksWithClaude(data) {
  const apiKey = getEnv('ANTHROPIC_API_KEY');
  const todayStr = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd (E)');

  const systemPrompt = `入力メッセージから受取人が実行すべき「明確なタスク」を抽出してください。
相対日付（「来週火曜」等）は基準日（今日: ${todayStr}）から厳密にYYYY-MM-DDへ変換してください。
メルマガやタスクがない連絡は has_task: false としてください。`;

  const tools = [{
    name: "submit_extracted_tasks",
    description: "抽出タスク配列の出力",
    input_schema: {
      type: "object",
      properties: {
        has_task: { type: "boolean" },
        tasks: {
          type: "array",
          items: {
            type: "object",
            properties: {
              task_name: { type: "string" },
              due_date: { type: ["string", "null"] },
              priority: { type: "string", enum: ["High", "Medium", "Low"] },
              summary: { type: "string" }
            },
            required: ["task_name", "due_date", "priority", "summary"]
          }
        }
      },
      required: ["has_task", "tasks"]
    }
  }];

  const payload = {
    model: "claude-3-5-sonnet-20241022",
    max_tokens: 1500,
    temperature: 0,
    system: systemPrompt,
    messages: [{ role: "user", content: `表題: ${data.title}\n本文:\n${data.content}` }],
    tools: tools,
    tool_choice: { type: "tool", name: "submit_extracted_tasks" }
  };

  const response = fetchWithBackoff("https://api.anthropic.com/v1/messages", {
    method: "post",
    contentType: "application/json",
    headers: { "x-api-key": apiKey, "anthropic-version": "2023-06-01" },
    payload: JSON.stringify(payload)
  });

  const resJson = JSON.parse(response.getContentText());
  const toolUse = resJson.content?.find(b => b.type === "tool_use");
  return toolUse ? toolUse.input : { has_task: false, tasks: [] };
}

function createNotionTask(task) {
  const notionApiKey = getEnv('NOTION_API_KEY');
  const databaseId = getEnv('NOTION_DATABASE_ID');

  // 重複チェック
  const queryUrl = `https://api.notion.com/v1/databases/${databaseId}/query`;
  const existing = fetchWithBackoff(queryUrl, {
    method: "post",
    contentType: "application/json",
    headers: { "Authorization": `Bearer ${notionApiKey}`, "Notion-Version": "2022-06-28" },
    payload: JSON.stringify({ filter: { property: "External ID", rich_text: { equals: task.externalId } } })
  });

  if (JSON.parse(existing.getContentText()).results?.length > 0) return;

  const properties = {
    "Task Name": { title: [{ text: { content: task.title } }] },
    "Status": { status: { name: "Not started" } },
    "Priority": { select: { name: task.priority } },
    "Source": { select: { name: task.sourceType } },
    "Sender": { rich_text: [{ text: { content: task.sender.substring(0, 100) } }] },
    "Summary": { rich_text: [{ text: { content: task.summary } }] },
    "External ID": { rich_text: [{ text: { content: task.externalId } }] },
    "Source URL": { url: task.sourceUrl }
  };
  if (task.dueDate) properties["Due Date"] = { date: { start: task.dueDate } };

  fetchWithBackoff("https://api.notion.com/v1/pages", {
    method: "post",
    contentType: "application/json",
    headers: { "Authorization": `Bearer ${notionApiKey}`, "Notion-Version": "2022-06-28" },
    payload: JSON.stringify({ parent: { database_id: databaseId }, properties: properties })
  });
}

function fetchWithBackoff(url, options, maxRetries = 4) {
  let attempt = 0, delay = 1000;
  while (attempt <= maxRetries) {
    try {
      const res = UrlFetchApp.fetch(url, Object.assign({}, options, { muteHttpExceptions: true }));
      const code = res.getResponseCode();
      if (code >= 200 && code < 300) return res;
      if (code === 429 || code >= 500) Utilities.sleep(delay);
      else throw new Error(`[HTTP ${code}] ${res.getContentText()}`);
    } catch (e) {
      if (attempt === maxRetries) throw e;
    }
    attempt++;
    delay *= 2;
  }
}

function sendSlackAlert(text) {
  const webhook = PropertiesService.getScriptProperties().getProperty('SLACK_WEBHOOK_URL');
  if (webhook) UrlFetchApp.fetch(webhook, { method: 'post', contentType: 'application/json', payload: JSON.stringify({ text }) });
}
